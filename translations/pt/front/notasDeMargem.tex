% A quebra respeita as palavras do identificador e conserva os seus caracteres.
\ExplSyntaxOn
\NewDocumentCommand{\codigoMargem}{m}{
  \tl_set:Nx \l_tmpa_tl {\tl_to_str:n {#1}}
  \regex_replace_all:nnN {([a-z0-9])([A-Z])} {\1\c{allowbreak}\2} \l_tmpa_tl
  \regex_replace_all:nnN {([/\x{5F}]|::)} {\1\c{allowbreak}} \l_tmpa_tl
  \tl_use:N \l_tmpa_tl
}
\ExplSyntaxOff
